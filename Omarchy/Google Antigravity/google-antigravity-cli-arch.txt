# ============================================================
# Google Antigravity CLI - Instalación en Arch Linux (Omarchy)
# ============================================================
#
# PREREQUISITOS:
#   sudo pacman -S curl bash
#
# NOTAS:
# - El instalador oficial (install.sh) descarga el binario a
#   $HOME/.local/bin, lo cual es compatible con Arch.
# - Si el instalador oficial falla por dependencias Debian,
#   se puede descargar el binario directamente desde:
#   https://antigravity.google/cli
# - Omarchy usa bash por defecto; si se usa otro shell, adaptar
#   la ruta del archivo de configuración (.zshrc, .fishrc, etc.)
# ============================================================

curl -fsSL https://antigravity.google/cli/install.sh | bash

*******************************

export PATH="$HOME/.local/bin:$PATH"
command -v agy
agy --version

*******************************

# Persistir PATH en el archivo de configuración del shell
# Para bash:
grep -qxF 'export PATH="$HOME/.local/bin:$PATH"' "$HOME/.bashrc" 2>/dev/null || printf '\nexport PATH="$HOME/.local/bin:$PATH"\n' >>"$HOME/.bashrc"

# Para zsh (si se usa en lugar de bash):
# grep -qxF 'export PATH="$HOME/.local/bin:$PATH"' "$HOME/.zshrc" 2>/dev/null || printf '\nexport PATH="$HOME/.local/bin:$PATH"\n' >>"$HOME/.zshrc"

*******************************

agy --help | sed -n '1,24p'

*******************************

mkdir -p "$HOME/.local/bin"

tee "$HOME/.local/bin/update-antigravity-cli" > /dev/null <<'EOF'
#!/usr/bin/env bash
set -euo pipefail

export PATH="$HOME/.local/bin:$PATH"

if [ ! -x "$HOME/.local/bin/agy" ]; then
	curl -fsSL https://antigravity.google/cli/install.sh | bash
else
	"$HOME/.local/bin/agy" update
fi

"$HOME/.local/bin/agy" --version
EOF

chmod +x "$HOME/.local/bin/update-antigravity-cli"

*******************************

update-antigravity-cli
